  add_filter( 'rest_endpoints', function ( $endpoints ) {
        if ( ! is_user_logged_in() ) {
                unset( $endpoints['/wp/v2/users'] );
                unset( $endpoints['/wp/v2/users/(?P<id>[\d]+)'] );
        }
        return $endpoints;
  } );

  // 2. /?author=1 → /author/xxx/ のリダイレクトを止める
  add_action( 'template_redirect', function () {
        if ( is_admin() ) {
                return;
        }
        if ( isset( $_GET['author'] ) || is_author() ) {
                wp_safe_redirect( home_url( '/' ), 301 );
                exit;
        }
  } );

  // 3. 著者アーカイブへのリンクを出力しない
  add_filter( 'author_link', function () {
        return home_url( '/' );
  }, 10, 1 );

  // 4. oEmbed レスポンスから著者情報を除去
  add_filter( 'oembed_response_data', function ( $data ) {
        unset( $data['author_name'], $data['author_url'] );
        return $data;
  } );

  // 5. xmlrpc.php を無効化
  add_filter( 'xmlrpc_enabled', '__return_false' );
  add_filter( 'wp_headers', function ( $headers ) {
        unset( $headers['X-Pingback'] );
        return $headers;
  } );
  remove_action( 'wp_head', 'rsd_link' );

  // 6. WordPress のバージョン表記を出さない
  remove_action( 'wp_head', 'wp_generator' );

  // 7. xmlrpc.php へのアクセス自体を遮断
  add_action( 'plugins_loaded', function () {

        }
  } );

### メールアドレス書き換え作業

old@example.comをnew@example.comに変更する。

`git-filter-repo`というモジュールを使用する。

作業の流れ

1. 環境構築
   - WSLに入りPython3がインストール済みのことを確認する
   - スクリプトを直接ダウンロードする
     ```
     mkdir -p /tmp/gfr
     curl -o /tmp/gfr/git-filter-repo https://raw.githubusercontent.com/newren/git-filter-repo/main/git-filter-repo
     ```
1. mailmapファイルを作成し、旧アドレスと新アドレスの対応を作成する
   ```
   ~/git-mailmap.txt
   username <new@example.com> <old@example.com>
   ```
1. リポジトリ一覧取得
   ```
   gh repo list username --limit 100 --json url,name -q '.[] | "\(.name) \(.url)"' > ~/work/repo-list.txt
   ```
1. 作業用ディレクトリ作成
   ```
   mkdir -p ~/work/backup
   mkdir -p ~/work/filter
   ```
1. スクリプト作成

   ```
   #!/bin/bash
   set -e

   SCRIPT="/tmp/gfr/git-filter-repo"
   MAILMAP="$HOME/git-mailmap.txt"
   REPO_LIST="$HOME/work/repo-list.txt"
   BACKUP_DIR="$HOME/work/backup"
   FILTER_DIR="$HOME/work/filter"

   while read -r name url; do
       echo "========================================"
       echo "Processing: $name"
       echo "========================================"

       # 既に処理済みならスキップ(再実行時の途中再開に対応)
       if [ -d "$FILTER_DIR/$name.git" ]; then
           echo "Already processed, skipping: $name"
           continue
       fi

       # 1. バックアップ(ミラークローン)
       echo "--- Backing up ---"
       git clone --mirror "$url" "$BACKUP_DIR/$name.git"

       # 2. filter-repo用のクローン
       echo "--- Cloning for filter ---"
       git clone --bare "$url" "$FILTER_DIR/$name.git"

       # 3. filter-repo実行 + push
       echo "--- Running filter-repo ---"
       (
           cd "$FILTER_DIR/$name.git"
           python3 "$SCRIPT" --mailmap "$MAILMAP" --force
           git remote add origin "$url"
           git push --mirror
       )

       echo "--- Done: $name ---"
       echo ""
   done < "$REPO_LIST"

   echo "All repositories processed."
   ```

1. 書き換え後の確認
   ```
   for dir in ~/work/filter/*.git; do
       name=$(basename "$dir" .git)
       emails=$(git -C "$dir" log --format='%ae' | sort -u)
       echo "=== $name ==="
       echo "$emails"
       echo ""
   done | tee ~/work/verify-all.log
   ```
1. ローカルリポジトリの更新
   ```
   git fetch origin
   git reset --hard origin/master
   git reflog expire --expire=now --all
   git gc --prune=now
   ```
